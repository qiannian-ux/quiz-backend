## 步骤 1：VPaySign.java（签名工具）

```java
package com.example.quiz.util;

import javax.crypto.Mac;
import javax.crypto.spec.SecretKeySpec;
import java.nio.charset.StandardCharsets;
import java.util.HexFormat;

public class VPaySign {

    /** HMAC-SHA256 → 十六进制小写（微信要求） */
    public static String hmacSha256(String key, String msg) throws Exception {
        Mac mac = Mac.getInstance("HmacSHA256");
        mac.init(new SecretKeySpec(key.getBytes(StandardCharsets.UTF_8), "HmacSHA256"));
        byte[] raw = mac.doFinal(msg.getBytes(StandardCharsets.UTF_8));
        return HexFormat.of().formatHex(raw);
    }

    /** 支付签名 paySig：C 端 uri 固定 requestVirtualPayment */
    public static String calcPaySig(String signData, String appKey) throws Exception {
        return hmacSha256(appKey, "requestVirtualPayment&" + signData);
    }

    /** 用户态签名 signature：msg 不带 uri 前缀，只有 signData 本身 */
    public static String calcSignature(String signData, String sessionKey) throws Exception {
        return hmacSha256(sessionKey, signData);
    }

    /** 查单签名 pay_sig：B 端 uri = /xpay/query_order */
    public static String calcQuerySig(String bodyJson, String appKey) throws Exception {
        return hmacSha256(appKey, "/xpay/query_order&" + bodyJson);
    }

    /**
     * ⚠️ 致命坑：signData 的 JSON 键顺序必须固定，禁止 Gson/Jackson 默认序列化（会按字母序重排）。
     * 这里手动拼，顺序照抄官方。
     */
    public static String buildSignData(String offerId, int buyQuantity, int env,
                                       String currencyType, String productId,
                                       int goodsPrice, String outTradeNo, String attach) {
        return "{\"offerId\":\"" + offerId + "\","
                + "\"buyQuantity\":" + buyQuantity + ","
                + "\"env\":" + env + ","
                + "\"currencyType\":\"" + currencyType + "\","
                + "\"productId\":\"" + productId + "\","
                + "\"goodsPrice\":" + goodsPrice + ","
                + "\"outTradeNo\":\"" + outTradeNo + "\","
                + "\"attach\":\"" + attach + "\"}";
    }
}
```

### 2.1 SQL（加到 schema-full.sql 或单独执行，IF NOT EXISTS）

```sql
CREATE TABLE IF NOT EXISTS pay_order (
  id           BIGINT PRIMARY KEY AUTO_INCREMENT,
  user_id      BIGINT NULL,
  openid       VARCHAR(64),
  out_trade_no VARCHAR(32) NOT NULL,
  wx_order_id  VARCHAR(64),
  product_id   VARCHAR(32),
  amount_fen   INT,
  status       TINYINT NOT NULL DEFAULT 0,   -- 0 待支付 1 已发货 2 已退款
  attach       VARCHAR(255),
  created_at   DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at   DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  UNIQUE KEY uk_out_trade_no (out_trade_no)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- premium 标记（发货时置 1）：在 user 表加一列
-- ⚠️ MySQL 不支持 `ADD COLUMN IF NOT EXISTS`（那是 MariaDB 语法），用存储过程做幂等判断
DROP PROCEDURE IF EXISTS add_premium_col;
DELIMITER $$
CREATE PROCEDURE add_premium_col()
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM information_schema.COLUMNS
    WHERE TABLE_SCHEMA = DATABASE()
      AND TABLE_NAME = 'user'
      AND COLUMN_NAME = 'premium'
  ) THEN
    ALTER TABLE `user` ADD COLUMN `premium` TINYINT NOT NULL DEFAULT 0;
  END IF;
END$$
DELIMITER ;
CALL add_premium_col();
DROP PROCEDURE IF EXISTS add_premium_col;
```

## 步骤 4：PayController.java

```java
package com.example.quiz.controller;

import ...;
import org.springframework.web.bind.annotation.*;
import java.util.Map;

@RestController
public class PayController {

    @Autowired
    private PayService payService;

    /** ① 前端下单拿签名（需登录，走 JWT） */
    @PostMapping("/api/pay/order")
    public Map<String, Object> order(@RequestBody OrderReq req,
                                     @RequestAttribute("userId") Long userId) {
        return payService.createOrder(userId, req.getProductId(), req.getAttach());
    }

    /** ③ 前端 / 定时任务兜底查单（需登录） */
    @PostMapping("/api/pay/query")
    public Map<String, Object> query(@RequestBody QueryReq req,
                                     @RequestAttribute("userId") Long userId) {
        return payService.queryOrder(req.getOutTradeNo(), req.getOpenid());
    }

    /** ② 微信发货推送（不走 JWT，拦截器白名单放行） */
    @PostMapping(value = "/pay/notify", produces = "text/xml;charset=UTF-8")
    public String notify(@RequestBody String xml) {
        return payService.handleNotify(xml);
    }

    // ---- DTO ----
    public static class OrderReq {
        private String productId;
        private String attach;
        // getter/setter
    }
    public static class QueryReq {
        private String outTradeNo;
        private String openid;
        // getter/setter
    }
}
```

> `@RequestAttribute("userId")` 的 key 名需和你 `JwtInterceptor` 里 `request.setAttribute(...)` 设的一致（通常是 `"userId"`）。

## 步骤 5：PayService.java（核心）

```java
package com.example.quiz.service;

import ...;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.*;
import org.springframework.stereotype.Service;
import org.springframework.web.client.RestClient;
import java.util.*;

@Service
public class PayService {

    @Value("${wx.vpay.app-id}")   private String appId;
    @Value("${wx.vpay.offer-id}") private String offerId;
    @Value("${wx.vpay.app-key}")  private String appKey;
    @Value("${wx.vpay.env}")      private int env;

    @Autowired private PayOrderMapper orderMapper;
    @Autowired private UserPlatformMapper userPlatformMapper; // 你已有
    @Autowired private UserMapper userMapper;                // 你已有（premium 字段）
    private final RestClient rest = RestClient.create();

    /** 道具价目（分），与微信后台【道具管理】严格一致 */
    private int priceOf(String productId) {
        if ("unlock_all".equals(productId)) return 590;
        if ("tip_milktea".equals(productId)) return 600;
        throw new IllegalArgumentException("未知道具: " + productId);
    }

    // ================= ① 下单 =================
    public Map<String, Object> createOrder(Long userId, String productId, String attach) {
        UserPlatform up = userPlatformMapper.selectByUserId(userId);
        if (up == null || up.getSessionKey() == null)
            throw new RuntimeException("请重新登录后再购买");

        int goodsPrice = priceOf(productId);
        String outTradeNo = "T" + System.currentTimeMillis()
                + String.format("%04d", new Random().nextInt(10000)); // 18位，T 开头，唯一

        String signData = VPaySign.buildSignData(
                offerId, 1, env, "CNY", productId, goodsPrice, outTradeNo,
                attach == null ? String.valueOf(userId) : attach);
        String paySig = VPaySign.calcPaySig(signData, appKey);
        String signature = VPaySign.calcSignature(signData, up.getSessionKey());

        // 落库
        PayOrder o = new PayOrder();
        o.setUserId(userId);
        o.setOpenid(up.getOpenid());
        o.setOutTradeNo(outTradeNo);
        o.setProductId(productId);
        o.setAmountFen(goodsPrice);
        o.setStatus(0);
        o.setAttach(signData); // attach 原样透传，这里也可只存 userId
        orderMapper.insert(o);

        // 返回（前端透传给 wx.requestVirtualPayment，前端不重算签名）
        Map<String, Object> payData = new LinkedHashMap<>();
        payData.put("offerId", offerId);
        payData.put("signData", signData);
        payData.put("paySig", paySig);
        payData.put("signature", signature);
        payData.put("env", env);
        payData.put("outTradeNo", outTradeNo);
        payData.put("mode", "short_series_goods");
        return payData;
    }

    // ================= ② 发货推送 =================
    public String handleNotify(String xml) {
        try {
            Map<String, String> m = parseXml(xml);
            if (!"xpay_goods_deliver_notify".equals(m.get("Event")))
                return errXml("事件不匹配");

            String outTradeNo = m.get("OutTradeNo");
            String wxOrderId  = m.get("MchOrderNo");
            String productId  = m.get("ProductId");
            String openid     = m.get("OpenId");

            // 幂等：同一 wx_order_id 只发一次货
            PayOrder byWx = orderMapper.selectByWxOrderId(wxOrderId);
            if (byWx != null && byWx.getStatus() == 1) return okXml();

            PayOrder order = orderMapper.selectByOutTradeNo(outTradeNo);
            if (order == null) return errXml("订单不存在");

            // 验签（§6 兜底，个人版推送字段以官方为准；先放开，幂等+HTTPS 顶上）
            if (!verifyPush(xml)) return errXml("验签失败");

            // 发货路由
            // 实现版不改动 User 实体：前端支付成功已写 storage.premium，
            // 后端以 pay_order.status=1 作为"已解锁"的服务器真相（跨端可查），无需 setPremium。
            if ("unlock_all".equals(productId)) {
                // 如需服务端真相，可在此加一个 /api/pay/status 接口按 user_id 查 pay_order
            }
            // tip_milktea：仅记打赏，不发货权益

            order.setWxOrderId(wxOrderId);
            order.setStatus(1);
            orderMapper.updateById(order);
            return okXml();
        } catch (Exception e) {
            return errXml(e.getMessage());
        }
    }

    // ================= ③ 兜底查单 =================
    public Map<String, Object> queryOrder(String outTradeNo, String openid) {
        String body = "{\"openid\":\"" + openid + "\",\"env\":" + env
                + ",\"order_id\":\"" + outTradeNo + "\"}";
        String paySig = VPaySign.calcQuerySig(body, appKey);

        // ⚠️ query_order 的鉴权方式以官方最新文档为准；
        //    下面是最可能的形态（body + pay_sig），若官方要求 access_token/特定 header 再补。
        Map<String, Object> req = new LinkedHashMap<>();
        req.put("openid", openid);
        req.put("env", env);
        req.put("order_id", outTradeNo);
        req.put("pay_sig", paySig);

        ResponseEntity<String> resp = rest.post()
                .uri("https://api.weixin.qq.com/xpay/query_order")
                .contentType(MediaType.APPLICATION_JSON)
                .body(req)
                .retrieve()
                .toEntity(String.class);

        // 解析 resp：若已支付但本地未发货，复用发货逻辑
        // （解析逻辑类似 handleNotify，按 wx_order_id 幂等，此处从简）
        return Map.of("raw", resp.getBody());
    }

    // -------- 小工具 --------
    private Map<String, String> parseXml(String xml) throws Exception {
        Map<String, String> map = new HashMap<>();
        DocumentBuilder db = DocumentBuilderFactory.newInstance().newDocumentBuilder();
        Document doc = db.parse(new ByteArrayInputStream(xml.getBytes(StandardCharsets.UTF_8)));
        NodeList list = doc.getElementsByTagName("*");
        for (int i = 0; i < list.getLength(); i++) {
            Element e = (Element) list.item(i);
            if (e.getChildNodes().getLength() == 1)
                map.put(e.getTagName(), e.getTextContent());
        }
        return map;
    }

    private boolean verifyPush(String xml) {
        // TODO：个人版发货推送的精确验签字段以官方为准。
        // 稳妥做法：① 若 XML 带 signature 字段用 AppKey 验（同 VPaySign 模式）
        //          ② 依赖【消息推送】Token 做来源校验
        //          ③ 至少 HTTPS + 幂等已挡住绝大多数伪造。
        // 联调阶段先返回 true，上线前补真实验签。
        return true;
    }

    private String okXml() {
        return "<xml><ErrCode>0</ErrCode><ErrMsg><![CDATA[success]]></ErrMsg></xml>";
    }
    private String errXml(String msg) {
        return "<xml><ErrCode>1</ErrCode><ErrMsg><![CDATA[" + msg + "]]></ErrMsg></xml>";
    }
}
```

    registry.addInterceptor(jwtInterceptor)
            .addPathPatterns("/api/**")

## 步骤 7：发货推送验签说明（不确定点，先知道）

官方 5.4.1 只写「服务器校验签名」，**没给推送体的签名算法**。所以 `verifyPush` 先返回 `true`，靠 **HTTPS + 幂等** 兜底。
上线前补：以微信 MP 后台实际推送字段为准实现验签（个人版可能是 XML 自带 `signature` 字段，按 AppKey 验）。幂等保证「重复通知也只发一次货」。

## 常见坑速查

- **签名对不上** → signData 键顺序被 Gson/Jackson 重排。必须用 `VPaySign.buildSignData` 手动拼。
- **报 -15005** → `session_key` 过期，让前端重新 `wx.login` 拿最新再下单。
- **-1003 / 道具价格不符** → `goodsPrice` 分 与微信后台道具价不一致。
- **微信一直重试** → `/pay/notify` 没返回 `ErrCode 0`，或没被白名单放行（401）。
- **outTradeNo 非法** → 必须 8–32 位、不能下划线开头、唯一不复用（本代码用 `T`+时间戳+随机，合规）。
- **金额单位** → 全程「分」，不换算；`env` 固定 `0`。
